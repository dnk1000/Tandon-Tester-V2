# @file CMakeLists.txt
# @brief Native host targets compiling production code against test doubles.
#
cmake_minimum_required(VERSION 3.16)
project(tm100_host_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
add_executable(safety_tests test_main.c ../main/drive.c ../main/config_values.c
    ../main/command.c ../main/timing.c ../main/power_values.c ../main/diagnostics.c)
target_include_directories(safety_tests PRIVATE stubs ../main)
target_compile_definitions(safety_tests PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(safety_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(safety_tests m)
add_test(NAME safety_tests COMMAND safety_tests)
add_executable(nvs_tests test_nvs.c ../main/config.c ../main/config_values.c)
target_include_directories(nvs_tests PRIVATE stubs ../main)
target_compile_options(nvs_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME nvs_tests COMMAND nvs_tests)
add_executable(hal_tests test_hal.c ../main/floppy_hal.c)
target_include_directories(hal_tests PRIVATE stubs ../main)
target_compile_options(hal_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME hal_tests COMMAND hal_tests)
add_executable(power_driver_tests test_power_driver.c ../main/power_values.c)
target_include_directories(power_driver_tests PRIVATE stubs ../main)
target_compile_options(power_driver_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME power_driver_tests COMMAND power_driver_tests)
